# Ad Search Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the Field Desk MCP a lane that searches Meta Ad Library and returns the *ads themselves* — copy, dates, and media URLs — so an MCP client can judge creative directly, without hunt's app/funnel vetting.

**Architecture:** Two admin MCP tools (`find_ads`, `ad_search_status`) queue and read a `pipeline_jobs` kind=`'ad_search'` job. A new Python executor pages Ad Library and writes every `AdRecord` verbatim into two new `frontend` tables. The existing hunt daemon gains a second lane to drain the new kind. Ranking (longest-running first, capped per advertiser) is applied at read time in Go, never at scrape time.

**Tech Stack:** Go 1.x + gorm + `github.com/modelcontextprotocol/go-sdk/mcp` (API/MCP), Python 3.13 + SQLAlchemy Core `text()` (pipeline), PostgreSQL, pytest, Go testing + testify.

**Spec:** `docs/superpowers/specs/2026-08-07-ad-search-design.md` — read it before Task 1.

## Global Constraints

- **Zero side effects on the discovery corpus.** An ad search must not write `discovery.candidates`, `public.advertisers`, `discovery.funnel_candidates`, `discovery.dynamic_keywords`, or `public.pipeline_jobs WHERE kind='enrich'`. Asserted as a mutation check in Task 2, re-asserted live in Task 6.
- **No media mirroring.** Store and return Meta's own `image_urls` / `video_urls`. Do not call `download_creative`, do not write `public.creatives`, do not add ffmpeg.
- **Product copy voice** (project CLAUDE.md): every user-facing string — tool descriptions, notes, error messages, progress events — is plain, clear, professional. No marketing words, no exclamation marks. Lowercase-first error messages ending in a period, matching `tools_engine.go` ("provide at least one term or one competitor.").
- **Python DB convention:** conn-first. Every helper takes an open connection; the caller owns the transaction. All SQL via `text()` with bound params.
- **Never hold a transaction across a network call.** Scrape outside `begin()`, persist inside a short one.
- **No bare `%` in migration files** — `apply_migrations` is placeholder-sensitive.
- **Test DB:** `AGF_TEST_DB_URL=postgresql+psycopg://agf:agf@localhost:55433/agf_it`. The zero-failed gate runs last and solo.
- **Commit directly to `main`.** One logical unit per commit.

```sql
-- 166_ad_searches.sql — the MCP ad-research lane (design 2026-08-07).
-- Ad searches keep the ADS that hunt discards: no triage, no vetoes, no writes
-- to the discovery corpus. Media URLs are Meta's own (fbcdn, signed, expiring);
-- nothing is mirrored, the MCP client fetches the bytes itself. Ranking is
-- read-time, so nothing derived is stored here — only what Meta returned.
CREATE TABLE IF NOT EXISTS frontend.ad_searches (
    id           BIGSERIAL PRIMARY KEY,
    created_by   TEXT NOT NULL,
    terms        JSONB NOT NULL DEFAULT '[]',
    competitors  JSONB NOT NULL DEFAULT '[]',
    settings     JSONB NOT NULL DEFAULT '{}',
    intent       TEXT,
    status       TEXT NOT NULL DEFAULT 'queued'
                 CHECK (status IN ('queued', 'running', 'done', 'error')),
    progress     JSONB NOT NULL DEFAULT '{}',
    error_text   TEXT,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
    started_at   TIMESTAMPTZ,
    finished_at  TIMESTAMPTZ
);

CREATE TABLE IF NOT EXISTS frontend.ad_search_results (
    id               BIGSERIAL PRIMARY KEY,
    search_id        BIGINT NOT NULL
                     REFERENCES frontend.ad_searches(id) ON DELETE CASCADE,
    ad_archive_id    TEXT NOT NULL,
    page_id          TEXT,
    page_name        TEXT,
    source_term      TEXT,
    is_active        BOOLEAN,
    start_date       TIMESTAMPTZ,
    end_date         TIMESTAMPTZ,
    collation_count  INT,
    page_like_count  BIGINT,
    cta_type         TEXT,
    cta_text         TEXT,
    link_url         TEXT,
    body_text        TEXT,
    image_urls       JSONB NOT NULL DEFAULT '[]',
    video_urls       JSONB NOT NULL DEFAULT '[]',
    first_seen       TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (search_id, ad_archive_id)
);

CREATE INDEX IF NOT EXISTS ix_ad_search_results_search
    ON frontend.ad_search_results (search_id, id);
```

```python
"""Repo for frontend.ad_search* — the MCP ad-research lane (design 2026-08-07).

Conn-first: every helper takes an open connection and the CALLER owns the txn
(same convention as db/hunts.py, db/pipeline_jobs.py). All SQL uses text() with
bound params.

Nothing derived is stored: days_running and the per-advertiser cap are computed
at read time in Go, so the same scrape can be re-cut with different knobs.
"""
import json

from sqlalchemy import text

# Per-search scrape knobs. Defaults mirror the executor constants; stored values
# overlay these, bounded below — the worker is the one place that decides what
# is legal, so Go passes values through unchecked.
DEFAULT_AD_SEARCH_SETTINGS = {"pages": 5, "listing_pages": 10}
_SETTINGS_BOUNDS = {"pages": (1, 20), "listing_pages": (1, 30)}


def clamp_ad_search_settings(raw: dict | None) -> dict:
    """Effective per-search knobs: defaults overlaid with bounded stored values."""
    out = dict(DEFAULT_AD_SEARCH_SETTINGS)
    for key, (lo, hi) in _SETTINGS_BOUNDS.items():
        val = (raw or {}).get(key)
        # bool is an int subclass: True would clamp to 1 page and quietly
        # cripple the search, so it is rejected as a non-number.
        if isinstance(val, (int, float)) and not isinstance(val, bool):
            out[key] = max(lo, min(hi, int(val)))
    return out


def load_search(conn, search_id: int):
    """One ad_searches row (terms/competitors as Python lists) or None."""
    return conn.execute(text(
        "SELECT id, terms, competitors, settings, intent, status "
        "FROM frontend.ad_searches WHERE id = :i"), {"i": search_id}).first()


def mark_search_running(conn, search_id: int) -> None:
    conn.execute(text(
        "UPDATE frontend.ad_searches "
        "SET status = 'running', started_at = now() WHERE id = :i"),
        {"i": search_id})


def finish_search(conn, search_id: int, *, status: str,
                  error: str | None = None) -> None:
    conn.execute(text(
        "UPDATE frontend.ad_searches "
        "SET status = :s, error_text = :e, finished_at = now() WHERE id = :i"),
        {"i": search_id, "s": status, "e": error})


def update_search_progress(conn, search_id: int, progress: dict) -> None:
    conn.execute(text(
        "UPDATE frontend.ad_searches SET progress = CAST(:p AS jsonb) "
        "WHERE id = :i"), {"i": search_id, "p": json.dumps(progress)})


def upsert_ad_result(conn, row: dict) -> bool:
    """One ad per search. A re-find (another term, a later page) refreshes the
    VOLATILE fields — is_active, end_date, collation_count, page_like_count and
    the media URLs, which are signed and rotate — and keeps source_term, which
    records the term that found the ad FIRST.

    start_date/end_date arrive from Meta as unix seconds; to_timestamp of a NULL
    param is NULL, so an undated ad stores NULL rather than the epoch.
    Returns True when the row was newly inserted (xmax = 0 on an insert).
    """
    return bool(conn.execute(text(
        "INSERT INTO frontend.ad_search_results "
        "  (search_id, ad_archive_id, page_id, page_name, source_term, "
        "   is_active, start_date, end_date, collation_count, page_like_count, "
        "   cta_type, cta_text, link_url, body_text, image_urls, video_urls) "
        "VALUES (:search_id, :ad_archive_id, :page_id, :page_name, :source_term, "
        "        :is_active, "
        "        to_timestamp(CAST(:start_date AS double precision)), "
        "        to_timestamp(CAST(:end_date AS double precision)), "
        "        :collation_count, :page_like_count, :cta_type, :cta_text, "
        "        :link_url, :body_text, CAST(:image_urls AS jsonb), "
        "        CAST(:video_urls AS jsonb)) "
        "ON CONFLICT (search_id, ad_archive_id) DO UPDATE SET "
        "  is_active = EXCLUDED.is_active, "
        "  end_date = EXCLUDED.end_date, "
        "  collation_count = EXCLUDED.collation_count, "
        "  page_like_count = EXCLUDED.page_like_count, "
        "  image_urls = EXCLUDED.image_urls, "
        "  video_urls = EXCLUDED.video_urls "
        "RETURNING (xmax = 0) AS inserted"), row).scalar())
```

```python
"""Ad search executor — the MCP research lane (design 2026-08-07).

Keeps the ADS that hunt discards: every AdRecord from every page persists to
frontend.ad_search_results verbatim. There is deliberately no classify_candidate,
no run_triage, no veto, no relevance judge, no enqueue_enrich and no keyword
mint here — an ad from a page with no store listing is often the most
instructive one, and vetting is what removed it.

Nothing derived is stored. Ranking (longest-running first, capped per
advertiser) happens at read time, so one scrape can be re-cut with different
knobs without paying for another.
"""
import asyncio
import collections
import json
import logging
import threading
import time
from datetime import datetime, timezone

from agf.db.ad_searches import (clamp_ad_search_settings, load_search,
                                mark_search_running, update_search_progress,
                                upsert_ad_result)
from agf.db.engine import begin
# Competitor disambiguation is imported, not copied: ranking typeahead rows by
# fuzzy name similarity was earned from a live finding (2026-07-15, 'Femmi' ->
# a fashion page at index 0), and a fork of it would silently rot.
from agf.discovery.hunt import (HUNT_COMPETITOR_ATTEMPTS, HUNT_COMPETITOR_CAP,
                                HUNT_COMPETITOR_COUNTRIES,
                                HUNT_COMPETITOR_TYPEAHEAD_TOP, HUNT_KW_DELAY,
                                HUNT_PAGE_DELAY, _name_similarity)
from agf.extract.errors import ExtractError

log = logging.getLogger("agf.ad_search")

AD_SEARCH_LANES = 3       # proven DataImpulse concurrency, same as hunt
AD_SEARCH_EVENT_CAP = 50


def _now_iso() -> str:
    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


class _Progress:
    """Thread-safe progress blob writer. Every mutation persists the whole
    locked state — short txn per write, no txn held across network calls.

    Not shared with hunt's _Progress: that one writes frontend.hunt_rounds, and
    parameterizing it would mean editing the hunt lane, which this lane does not
    touch."""

    def __init__(self, engine, search_id: int, terms_total: int):
        self._engine = engine
        self._search_id = search_id
        self._lock = threading.Lock()
        self._state = {"lanes": {}, "terms_done": 0, "terms_total": terms_total,
                       "found": 0, "new": 0, "events": [], "terms": {}}

    def _flush(self) -> None:
        with begin(self._engine) as conn:
            update_search_progress(conn, self._search_id, self._state)

    def event(self, msg: str) -> None:
        with self._lock:
            self._state["events"] = (
                self._state["events"] + [{"t": _now_iso(), "msg": msg}]
            )[-AD_SEARCH_EVENT_CAP:]
            self._flush()

    def mark_partial(self, msg: str) -> None:
        """Flag a search whose scrape stopped at the page budget. The flag lives
        IN self._state so every later lane flush preserves it."""
        with self._lock:
            self._state["partial"] = True
            self._state["events"] = (
                self._state["events"] + [{"t": _now_iso(), "msg": msg}]
            )[-AD_SEARCH_EVENT_CAP:]
            self._flush()

    def page(self, lane: int, term: str, depth: int, found: int, new: int) -> None:
        with self._lock:
            self._state["lanes"][str(lane)] = {"term": term, "page": depth}
            self._state["found"] += found
            self._state["new"] += new
            self._state["events"] = (self._state["events"] + [{
                "t": _now_iso(),
                "msg": f"Searching '{term}' — page {depth} · {found} ads"}]
            )[-AD_SEARCH_EVENT_CAP:]
            self._flush()

    def term_done(self, lane: int) -> None:
        with self._lock:
            self._state["terms_done"] += 1
            self._state["lanes"].pop(str(lane), None)
            self._flush()

    def term_stats(self, term: str, pages: int, found: int, new: int,
                   stop: str) -> None:
        with self._lock:
            self._state.setdefault("terms", {})[term] = {
                "pages": pages, "found": found, "new": new, "stop": stop}
            self._flush()


def _row_from(record, search_id: int, term: str) -> dict:
    """One AdRecord as an ad_search_results row. Everything Meta gave us is kept
    verbatim; nothing is classified, scored or judged."""
    return {
        "search_id": search_id,
        "ad_archive_id": record.ad_archive_id,
        "page_id": record.page_id,
        "page_name": record.page_name,
        "source_term": term,
        "is_active": record.is_active,
        "start_date": record.start_date,
        "end_date": record.end_date,
        "collation_count": record.collation_count,
        "page_like_count": record.page_like_count,
        "cta_type": record.cta_type,
        "cta_text": record.cta_text,
        "link_url": record.link_url,
        "body_text": record.body_text,
        "image_urls": json.dumps(list(record.image_urls or [])),
        "video_urls": json.dumps(list(record.video_urls or [])),
    }


def _persist_page(engine, search_id: int, term: str, records, stats: dict,
                  lock: threading.Lock) -> tuple[int, int]:
    """Upsert one page of ads in its own short txn — durable the moment it lands.
    Rows are ordered by ad id so concurrent lanes finding the same ads in
    opposite orders cannot deadlock (same discipline as hunt._persist_page).
    Returns (found, new)."""
    rows, seen = [], set()
    for record in records:
        if not record.ad_archive_id or record.ad_archive_id in seen:
            continue
        seen.add(record.ad_archive_id)
        rows.append(_row_from(record, search_id, term))
    if not rows:
        return 0, 0
    rows.sort(key=lambda r: r["ad_archive_id"])
    found = new = 0
    with begin(engine) as conn:
        for row in rows:
            new += int(upsert_ad_result(conn, row))
            found += 1
    with lock:
        stats["found"] += found
        stats["new"] += new
    return found, new


def _search_term(engine, extractor, search_id: int, lane: int, term: str,
                 progress: _Progress, stats: dict, lock: threading.Lock,
                 sleep, pages: int) -> tuple[int, int, int, str]:
    """Page one term to `pages`. Returns (pages_fetched, found, new, stop) where
    stop is 'exhausted' (Meta has no next page), 'cap' (the pages knob) or
    'error' (the search failed)."""
    cursor = None
    depth = 0
    term_found = term_new = 0
    while True:
        depth += 1
        try:
            records, cursor, has_next = asyncio.run(extractor.search_page(
                query=term, countries=["US"],
                search_type="KEYWORD_UNORDERED", cursor=cursor))
        except ExtractError:
            log.warning("ad search %d: search failed for %r", search_id, term)
            with lock:
                stats["errors"] += 1
            progress.event(f"Could not search '{term}' — skipped after an error")
            return depth - 1, term_found, term_new, "error"
        with lock:
            stats["pages"] += 1
        found, new = _persist_page(engine, search_id, term, records, stats, lock)
        term_found += found
        term_new += new
        progress.page(lane, term, depth, found, new)
        if not has_next or not cursor:
            return depth, term_found, term_new, "exhausted"
        if depth >= pages:
            return depth, term_found, term_new, "cap"
        sleep(HUNT_PAGE_DELAY)


def _pull_competitor(engine, extractor, search_id: int, name: str,
                     progress: _Progress, stats: dict, lock: threading.Lock,
                     listing_pages: int) -> None:
    """Resolve one competitor name to its Ad Library page and pull its ads.
    Typeahead matches page NAME and ranks big pages first, so the top hit is
    often a look-alike; rank the top rows by fuzzy similarity (likes break ties)
    and try the best few until one yields ads. Pulled GLOBALLY — niche apps
    often advertise only in EU/AU."""
    key = f"page:{name}"
    progress.event(f"Resolving {name}'s ad page")
    try:
        pages = asyncio.run(extractor.typeahead(query=name))
    except ExtractError:
        progress.event(f"Could not resolve {name} — skipped")
        with lock:
            stats["errors"] += 1
        progress.term_stats(key, 0, 0, 0, "error")
        return
    ranked = sorted(
        pages[:HUNT_COMPETITOR_TYPEAHEAD_TOP],
        key=lambda p: (_name_similarity(name, p.get("page_name")),
                       p.get("page_like_count") or 0),
        reverse=True)
    if not ranked:
        progress.event(f"No Ad Library page found for {name}")
        progress.term_stats(key, 0, 0, 0, "no_match")
        return
    attempts = 0
    for page in ranked[:HUNT_COMPETITOR_ATTEMPTS]:
        label = page.get("page_name") or name
        progress.event(f"Pulling ads from '{label}'")
        try:
            records, _truncated = asyncio.run(extractor.search_listing(
                page_id=page["page_id"],
                countries=HUNT_COMPETITOR_COUNTRIES,
                max_pages=listing_pages))
        except ExtractError:
            with lock:
                stats["errors"] += 1
            progress.term_stats(key, attempts, 0, 0, "error")
            return
        attempts += 1
        found, new = _persist_page(engine, search_id, key, records, stats, lock)
        if found:
            with lock:
                stats["competitor_pages"] += 1
            progress.page(0, key, 1, found, new)
            progress.term_stats(key, attempts, found, new, "listing_ok")
            return
    progress.term_stats(key, attempts, 0, 0, "no_results")


def run_ad_search(engine, extractors, search_id: int, *,
                  sleep=time.sleep) -> dict:
    """Execute one ad search end to end. Every page persists in its own txn, so
    a mid-scrape timeout keeps whatever already committed. Terminal status for
    the search row and the job is stamped by the caller (ad_search_one)."""
    with begin(engine) as conn:
        search = load_search(conn, search_id)
        if search is None:
            raise ValueError(f"ad search {search_id} not found")
        mark_search_running(conn, search_id)
    terms = list(search.terms or [])
    knobs = clamp_ad_search_settings(search.settings)
    stats = {"terms_searched": 0, "pages": 0, "found": 0, "new": 0,
             "errors": 0, "competitor_pages": 0}
    lock = threading.Lock()
    progress = _Progress(engine, search_id, terms_total=len(terms))
    progress.event(f"Ad search started — {len(terms)} terms")

    try:
        # competitors is a jsonb array of STRINGS as the Go tool writes it;
        # legacy [{"name": ...}] dicts are accepted for parity with hunt.
        raw = search.competitors or []
        competitors = [(c.get("name", "") if isinstance(c, dict) else str(c))
                       for c in raw]
        competitors = [c for c in competitors if c]
        for name in competitors[:HUNT_COMPETITOR_CAP]:
            _pull_competitor(engine, extractors[0], search_id, name, progress,
                             stats, lock, knobs["listing_pages"])

        queue = collections.deque(terms)

        def lane(lane_id: int) -> None:
            extractor = extractors[lane_id % len(extractors)]
            while True:
                try:
                    term = queue.popleft()
                except IndexError:
                    return
                pages, found, new, stop = _search_term(
                    engine, extractor, search_id, lane_id, term, progress,
                    stats, lock, sleep, knobs["pages"])
                progress.term_stats(term, pages, found, new, stop)
                with lock:
                    stats["terms_searched"] += 1
                progress.term_done(lane_id)
                sleep(HUNT_KW_DELAY)

        lanes = min(AD_SEARCH_LANES, max(1, len(terms)), max(1, len(extractors)))
        threads = [threading.Thread(target=lane, args=(i,), daemon=True)
                   for i in range(lanes)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
    except BaseException as exc:  # noqa: BLE001
        # The scrape budget's SIGALRM fires in THIS thread while lane threads may
        # still be flushing, so the partial marker goes INSIDE progress._state
        # where later flushes preserve it. A real failure gets no marker.
        from agf.discovery.ad_search_one import AdSearchTimeout
        if isinstance(exc, AdSearchTimeout):
            progress.mark_partial(
                "Stopped at the page budget — ads found so far are saved.")
        raise
    progress.event(f"Ad search finished — {stats['found']} ads")
    return stats
```

```python
"""Run ONE ad search to completion — the ad_search daemon lane's subprocess body.
Owns terminal status for both the search row and the pipeline job on every exit
path (the daemon's safety net only catches a killed process).

Unlike hunt_one there is a single budget: an ad search has no phase 5 to protect,
so there is nothing to keep alive after the scrape window closes."""
import argparse
import logging
import signal
import sys

from agf.db.ad_searches import finish_search
from agf.db.engine import begin, get_engine
from agf.db.pipeline_jobs import set_job_status
from agf.discovery.ad_search import AD_SEARCH_LANES, run_ad_search

log = logging.getLogger("agf.ad_search_one")


class AdSearchTimeout(BaseException):
    """SIGALRM watchdog — BaseException so no except-Exception swallows it."""


def _build_extractors(k: int):
    from agf.extract.facade import Extractor
    from agf.extract.proxy import ProxyPool
    proxies = ProxyPool.list_from_env()
    k = max(1, k)
    if not proxies:
        return [Extractor(proxies=ProxyPool(proxies))]
    if len(proxies) >= k:
        slices = [proxies[i::k] for i in range(k)]
        return [Extractor(proxies=ProxyPool(sl)) for sl in slices]
    return [Extractor(proxies=ProxyPool(proxies)) for _ in range(k)]


def main(argv=None) -> int:
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(name)s %(message)s")
    ap = argparse.ArgumentParser()
    ap.add_argument("--search-id", type=int, required=True)
    ap.add_argument("--job-id", type=int, required=True)
    ap.add_argument("--timeout", type=int, default=1800)
    args = ap.parse_args(argv)

    def _alarm(signum, frame):
        raise AdSearchTimeout(f"ad search exceeded {args.timeout}s")

    signal.signal(signal.SIGALRM, _alarm)
    engine = get_engine()
    try:
        extractors = _build_extractors(AD_SEARCH_LANES)
        signal.alarm(args.timeout)
        try:
            stats = run_ad_search(engine, extractors, args.search_id)
        except AdSearchTimeout as e:
            # The scrape didn't finish, but every persisted page is real and the
            # partial marker is already in progress. Commit 'done', never 'error'
            # — the ads that landed are exactly as usable as a full run's.
            signal.alarm(0)
            log.warning("ad search %d hit the page budget: %s", args.search_id, e)
            with begin(engine) as conn:
                finish_search(conn, args.search_id, status="done")
                set_job_status(conn, args.job_id, "done")
            return 0
        signal.alarm(0)
        with begin(engine) as conn:
            finish_search(conn, args.search_id, status="done")
            set_job_status(conn, args.job_id, "done")
        log.info("ad search %d done: %s", args.search_id, stats)
        return 0
    except AdSearchTimeout as e:      # a stray alarm outside the window
        _fail(engine, args, str(e))
        return 1
    except Exception as e:  # noqa: BLE001 — terminal-status parity on all paths
        log.exception("ad search %d failed", args.search_id)
        _fail(engine, args, str(e)[:500])
        return 1
    finally:
        signal.alarm(0)


def _fail(engine, args, msg: str) -> None:
    with begin(engine) as conn:
        finish_search(conn, args.search_id, status="error", error=msg)
        set_job_status(conn, args.job_id, "error", error=msg)


if __name__ == "__main__":
    sys.exit(main())
```

```go
// The scrape stores only what Meta returned; every derived number here
// (days_running, the per-advertiser cap) is computed at read time, so one
// scrape can be re-cut with different knobs without paying for another.
package repositories

import (
	"context"
	"encoding/json"

	"gorm.io/gorm"

	"github.com/edaslakavicius/emerging-app-finder/backend/db"
	"github.com/edaslakavicius/emerging-app-finder/backend/models"
)

// InsertAdSearch writes the search row and its pipeline job in ONE transaction,
// the same shape as InsertHuntRoundTx: a search with no job would sit queued
// forever, and a job with no search would kill the worker on its first read.
func InsertAdSearch(ctx context.Context, g *gorm.DB, createdBy string,
	terms, competitors, settings json.RawMessage, intent *string) (int, error) {
	var searchID int
	err := db.WithWriteTx(ctx, g, func(tx *gorm.DB) error {
		if e := tx.Raw(
			`INSERT INTO frontend.ad_searches (created_by, terms, competitors, settings, intent)
			 VALUES (?, CAST(? AS jsonb), CAST(? AS jsonb), CAST(? AS jsonb), ?) RETURNING id`,
			// terms/competitors are NOT NULL jsonb arrays and settings a NOT NULL
			// jsonb object: an empty RawMessage defaults to '[]'/'{}'. Go marshals a
			// nil slice to `null`, and a jsonb null is not a SQL NULL, so NOT NULL
			// would not catch it — it would reach the worker and die on list(...).
			createdBy, jsonbArgDefault(terms), jsonbArgDefault(competitors),
			jsonbArgDefaultObj(settings), intent,
		).Scan(&searchID).Error; e != nil {
			return e
		}
		return tx.Exec(
			`INSERT INTO public.pipeline_jobs (kind, advertiser_id)
			 VALUES ('ad_search', ?)
			 ON CONFLICT (kind, advertiser_id) WHERE status IN ('queued','running')
			 DO NOTHING`,
			searchID,
		).Error
	})
	return searchID, err
}

// AdSearchReportFor returns the status digest, or nil when the id is unknown.
func AdSearchReportFor(ctx context.Context, g *gorm.DB, searchID int) (*models.AdSearchReport, error) {
	var rows []models.AdSearchReport
	err := db.WithReadTx(ctx, g, func(tx *gorm.DB) error {
		return tx.Raw(
			`SELECT s.id, s.status, s.created_by, s.intent, s.terms, s.competitors,
			        s.settings, s.progress, s.created_at, s.finished_at,
			        (SELECT count(*) FROM frontend.ad_search_results r
			          WHERE r.search_id = s.id) AS total_ads
			   FROM frontend.ad_searches s
			  WHERE s.id = ?`, searchID).Scan(&rows).Error
	})
	if err != nil || len(rows) == 0 {
		return nil, err
	}
	return &rows[0], nil
}

// AdSearchResultsParams are the read-time knobs. Media is "any", "video" or
// "image"; the caller has already validated it.
type AdSearchResultsParams struct {
	SearchID      int
	MinDays       int
	Media         string
	PerAdvertiser int
	Limit         int
	Offset        int
}

// AdSearchResults ranks one search's ads longest-running first, capped per
// advertiser.
//
// The filters run BEFORE the window function on purpose. Capping first would
// let an advertiser's three longest-running IMAGE ads consume its whole quota
// and then be dropped by media='video', returning nothing from a page that does
// run video ads.
//
// days_running measures an active ad to now() and a finished one to its end
// date. GREATEST(0, ...) absorbs the occasional Meta row whose start date is in
// the future; an ad with no start date sorts last with 0.
func AdSearchResults(ctx context.Context, g *gorm.DB, p AdSearchResultsParams) ([]models.AdSearchRow, error) {
	var rows []models.AdSearchRow
	err := db.WithReadTx(ctx, g, func(tx *gorm.DB) error {
		return tx.Raw(
			`WITH scored AS (
			   SELECT r.*,
			          GREATEST(0, COALESCE(EXTRACT(day FROM
			            (COALESCE(r.end_date, now()) - r.start_date))::int, 0)) AS days_running
			     FROM frontend.ad_search_results r
			    WHERE r.search_id = ?
			 ), qualified AS (
			   SELECT * FROM scored
			    WHERE days_running >= ?
			      AND (? = 'any'
			           OR (? = 'video' AND jsonb_array_length(video_urls) > 0)
			           OR (? = 'image' AND jsonb_array_length(image_urls) > 0))
			 ), capped AS (
			   SELECT *, ROW_NUMBER() OVER (
			            PARTITION BY page_id ORDER BY days_running DESC, id) AS rn
			     FROM qualified
			 )
			 SELECT ad_archive_id, page_id, page_name, source_term, days_running,
			        is_active, collation_count, page_like_count, cta_text,
			        link_url, body_text, image_urls, video_urls
			   FROM capped
			  WHERE rn <= ?
			  ORDER BY days_running DESC, id
			  LIMIT ? OFFSET ?`,
			p.SearchID, p.MinDays, p.Media, p.Media, p.Media,
			p.PerAdvertiser, p.Limit, p.Offset).Scan(&rows).Error
	})
	return rows, err
}
```

```go
// Package-local: the ad-research lane. find_ads queues an Ad Library scrape that
// keeps the ADS (hunt keeps advertisers and discards the ads), and
// ad_search_status reads them back ranked longest-running first.
//
// Both are admin-only and re-check the role in the handler, the same defense in
// depth as tools_engine.go.
package mcpserver

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"

	"github.com/modelcontextprotocol/go-sdk/mcp"

	"github.com/edaslakavicius/emerging-app-finder/backend/db"
	"github.com/edaslakavicius/emerging-app-finder/backend/helpers"
	"github.com/edaslakavicius/emerging-app-finder/backend/repositories"
)

// adSearchSettings mirrors the per-search knobs the Python executor reads.
// Bounds are enforced by clamp_ad_search_settings on the worker side (pages
// 1-20, listing_pages 1-30); Go passes values through so there is exactly one
// place that decides what is legal. omitempty matters: an unset knob must be
// ABSENT, because a stored zero clamps up to the bound's minimum.
type adSearchSettings struct {
	Pages        int `json:"pages,omitempty"`
	ListingPages int `json:"listing_pages,omitempty"`
}

type findAdsIn struct {
	Terms        []string `json:"terms,omitempty" jsonschema:"keyword searches, US only"`
	Competitors  []string `json:"competitors,omitempty" jsonschema:"app or brand names; each is resolved to its Ad Library page and its ads are pulled globally"`
	Pages        int      `json:"pages,omitempty" jsonschema:"pages to read per term, 1-20 (default 5)"`
	ListingPages int      `json:"listing_pages,omitempty" jsonschema:"pages pulled per competitor page listing, 1-30 (default 10)"`
	Intent       string   `json:"intent,omitempty" jsonschema:"what this search is looking for; recorded on the search"`
}

func (h *handlers) findAds(ctx context.Context, _ *mcp.CallToolRequest, in findAdsIn) (*mcp.CallToolResult, any, error) {
	if h.u.Role != "admin" {
		return nil, nil, errors.New("This tool requires an admin account.")
	}
	// Same normalization and bounds as the hunt path, from the one place that
	// defines them: terms are search keys, so they are lowercased, trimmed and
	// deduped; competitors keep their case. Both helpers return a non-nil empty
	// slice, which is what keeps a nil slice from marshalling to `null` and
	// reaching the worker as a jsonb null that NOT NULL does not catch.
	in.Terms = helpers.NormalizeHuntTerms(in.Terms)
	in.Competitors = helpers.TrimHuntCompetitors(in.Competitors)

	if len(in.Terms) == 0 && len(in.Competitors) == 0 {
		return nil, nil, errors.New("provide at least one term or one competitor.")
	}
	// A search is real spend and the worker consumes every term it is handed, so
	// the caps are enforced rather than silently truncated.
	if len(in.Terms) > helpers.MaxHuntTerms {
		return nil, nil, fmt.Errorf("at most %d terms per search.", helpers.MaxHuntTerms)
	}
	if len(in.Competitors) > helpers.MaxHuntCompetitors {
		return nil, nil, fmt.Errorf("at most %d competitors per search.", helpers.MaxHuntCompetitors)
	}

	terms, err := json.Marshal(in.Terms)
	if err != nil {
		return nil, nil, errors.New("terms could not be encoded.")
	}
	competitors, err := json.Marshal(in.Competitors)
	if err != nil {
		return nil, nil, errors.New("competitors could not be encoded.")
	}
	settings, err := json.Marshal(adSearchSettings{
		Pages: in.Pages, ListingPages: in.ListingPages})
	if err != nil {
		return nil, nil, errors.New("settings could not be encoded.")
	}

	var intent *string
	if in.Intent != "" {
		intent = &in.Intent
	}
	searchID, err := repositories.InsertAdSearch(ctx, db.DB(), h.u.Email,
		terms, competitors, settings, intent)
	if err != nil {
		return nil, nil, errors.New("something went wrong queueing that search.")
	}
	return nil, map[string]any{
		"search_id": searchID,
		"status":    "queued",
		"note":      "The worker picks this up on its next poll. Read the ads with ad_search_status.",
	}, nil
}

// ---- ad_search_status ----

// adResultsDefaultLimit is smaller than the shared clampLimit default: an ad row
// carries two URL arrays and the full body text, so a page of them is far larger
// than a page of gem rows.
const (
	adResultsDefaultLimit  = 25
	adResultsMaxLimit      = 100
	adDefaultPerAdvertiser = 3
	adMaxPerAdvertiser     = 50
	adLibraryURLPrefix     = "https://www.facebook.com/ads/library/?id="
	// adMediaStaleHours is how long Meta's signed CDN links can be relied on.
	// Past this the ad facts are still true and only the media links have
	// rotted, so this is a note and never an error.
	adMediaStaleHours = 6
)

type adSearchStatusIn struct {
	SearchID      int    `json:"search_id" jsonschema:"the search to read"`
	MinDays       int    `json:"min_days,omitempty" jsonschema:"only ads running at least this many days"`
	Media         string `json:"media,omitempty" jsonschema:"any (default), video, or image"`
	PerAdvertiser int    `json:"per_advertiser,omitempty" jsonschema:"max ads from one advertiser, 1-50 (default 3)"`
	Limit         int    `json:"limit,omitempty" jsonschema:"max ads to return (default 25, max 100)"`
	Offset        int    `json:"offset,omitempty" jsonschema:"ads to skip for paging"`
}

func (h *handlers) adSearchStatus(ctx context.Context, _ *mcp.CallToolRequest, in adSearchStatusIn) (*mcp.CallToolResult, any, error) {
	if h.u.Role != "admin" {
		return nil, nil, errors.New("This tool requires an admin account.")
	}
	if in.SearchID <= 0 {
		return nil, nil, errors.New("search_id is required.")
	}
	media := in.Media
	switch media {
	case "", "any":
		media = "any"
	case "video", "image":
	default:
		return nil, nil, errors.New("media must be any, video, or image.")
	}
	report, err := repositories.AdSearchReportFor(ctx, db.DB(), in.SearchID)
	if err != nil {
		return nil, nil, errors.New("something went wrong reading that search.")
	}
	if report == nil {
		return nil, nil, errors.New("ad search not found.")
	}

	limit := in.Limit
	if limit <= 0 || limit > adResultsMaxLimit {
		limit = adResultsDefaultLimit
	}
	perAdvertiser := in.PerAdvertiser
	if perAdvertiser <= 0 || perAdvertiser > adMaxPerAdvertiser {
		perAdvertiser = adDefaultPerAdvertiser
	}
	minDays := in.MinDays
	if minDays < 0 {
		minDays = 0
	}
	offset := in.Offset
	if offset < 0 {
		offset = 0
	}

	ads, err := repositories.AdSearchResults(ctx, db.DB(), repositories.AdSearchResultsParams{
		SearchID:      in.SearchID,
		MinDays:       minDays,
		Media:         media,
		PerAdvertiser: perAdvertiser,
		Limit:         limit,
		Offset:        offset,
	})
	if err != nil {
		return nil, nil, errors.New("something went wrong reading that search's ads.")
	}
	for i := range ads {
		ads[i].LibraryURL = adLibraryURLPrefix + ads[i].AdArchiveID
	}

	out := map[string]any{
		"search_id": report.ID,
		"status":    report.Status,
		"search":    report,
		"total_ads": report.TotalAds,
		"ads":       ads,
		"limit":     limit,
		"offset":    offset,
	}
	notes := []string{}
	if len(ads) == limit {
		notes = append(notes, fmt.Sprintf(
			"Showing ads %d-%d. There may be more; call again with offset %d.",
			offset+1, offset+len(ads), offset+limit))
	}
	// Media URLs are Meta's own signed links and they expire. Say so rather than
	// letting a caller read a download failure as a missing creative.
	if time.Since(report.CreatedAt) > adMediaStaleHours*time.Hour {
		notes = append(notes, "This search is more than 6 hours old, so its image and video URLs may have expired. The ad text and dates are still accurate; run find_ads again for fresh media links.")
	}
	if len(notes) > 0 {
		out["note"] = strings.Join(notes, " ")
	}
	return nil, out, nil
}
```
