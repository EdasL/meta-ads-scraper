```python
def traction_delta(points, min_span_days: int = 7):
    """(raw_delta, per_day_rate) between earliest/latest non-None values of an
    ordered [(date, value)] series, or None when <2 usable points or the span
    is under min_span_days (too narrow to be a trend)."""
    vals = [(d, v) for d, v in points if v is not None]
    if len(vals) < 2:
        return None
    (d0, v0), (d1, v1) = vals[0], vals[-1]
    span = (d1 - d0).days
    if span < min_span_days:
        return None
    return (v1 - v0, (v1 - v0) / span)


_MAX_LAUNCH_AGE_S = 5 * 365 * 86400  # ignore garbage start_dates older than ~5y


def weekly_launch_counts(start_epochs, now_epoch: int):
    """Ad launches per week (zero-filled) from the earliest sane launch to
    now_epoch, oldest -> newest. Empty input -> []."""
    week = 7 * 86400
    eps = [e for e in start_epochs if e and now_epoch - e <= _MAX_LAUNCH_AGE_S]
    if not eps:
        return []
    first = min(eps)
    counts = [0] * (int((now_epoch - first) // week) + 1)
    for e in eps:
        counts[int((e - first) // week)] += 1
    return counts
```

              "longevity": 0.15, "iteration": 0.20}
D_WEIGHTS = {"installs": 0.55, "reviews": 0.45}

def demand(components):     return weighted_axis(components, D_WEIGHTS)
def whitespace(components): return weighted_axis(components, W_WEIGHTS)

def gem_score(sv, ur, fi, d=None, w=None, floor: float = 0.5) -> float:
    a = floor if sv is None else clamp01(sv)
    b = floor if ur is None else clamp01(ur)
    c = floor if fi is None else clamp01(fi)
    e = floor if d is None else clamp01(d)
    f = floor if w is None else clamp01(w)
    return 100.0 * (a ** 0.40) * (b ** 0.25) * (c ** 0.05) * (e ** 0.20) * (f ** 0.10)
