# Live-Adapting Review Queue Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The /review queue re-ranks live from the user's ratings — each rating steers the next cards served (TikTok-style), with no dependency on the offline `score_content_preferences.py` job.

**Architecture:** Port the Python `ContentPreferenceModel` (like−dislike direction + Platt calibration) into `backend/internal/reco`; `PreferenceQueue` fits it per request from `frontend.app_ratings` × `frontend.app_embeddings` (embeddings held in a 10-minute in-process cache) and scores all unrated matview candidates before the existing `assembleQueue` blend. The web deck shrinks from a one-shot 2,000-card fetch to a ~12-card buffer with a debounced single-flight background refill after each rating; refills replace the unseen tail and carry an `exclude` id list.

**Tech Stack:** Go 1.25 (Gin + GORM raw SQL, `backend/`), Next.js 16/React (`web/`), Postgres (matview + REAL[] embeddings), `go test` (+ `AGF_IT_DB_URL` integration), `node --test` for web pure-logic tests.

**Spec:** `docs/superpowers/specs/2026-07-06-live-review-queue-design.md`

```go
func unitVec(v []float64) []float64 {
	var n float64
	for _, x := range v {
		n += x * x
	}
	n = math.Sqrt(n)
	out := make([]float64, len(v))
	for i, x := range v {
		out[i] = x / n
	}
	return out
}
```

```go
package reco

import (
	"fmt"
	"math"
	"strconv"
	"strings"
)

// ContentModel is the Go port of src/agf/scoring/content_preference.py's
// ContentPreferenceModel: direction = l2norm(mean(liked) − mean(disliked));
// score(app) = dot(app, direction) (vectors are unit-norm, so cosine == dot);
// p_like is a Platt-style 1-D calibration of the raw score against the labels;
// why names the nearest liked app. Pure — no DB. The Python model stays the
// canonical reference; content_model_test.go pins parity with golden vectors.
type ContentModel struct {
	version    string
	dir        []float64
	a, b       float64
	likedVecs  [][]float64
	likedNames []string
}

// MinLikes is the coldstart threshold: fewer liked apps (with embeddings) than
// this and the model refuses to fit a direction (mirrors Python MIN_LIKES).
const MinLikes = 8

// LabeledVec is one rated app's content embedding plus its display name (used
// only for the "similar to <name>" why-string of liked apps).
type LabeledVec struct {
	Name string
	Vec  []float64
}

func l2norm(v []float64) []float64 {
	var n float64
	for _, x := range v {
		n += x * x
	}
	n = math.Sqrt(n)
	if n < 1e-12 {
		return v
	}
	out := make([]float64, len(v))
	for i, x := range v {
		out[i] = x / n
	}
	return out
}

// fitPlatt is the tiny gradient-descent logistic of label ~ a*raw + b, ported
// verbatim (500 iterations, lr 0.5, init a=1 b=0, non-finite → identity).
func fitPlatt(raws []float64, labels []float64) (float64, float64) {
	a, b := 1.0, 0.0
	n := float64(len(raws))
	if n < 1 {
		n = 1
	}
	const iters, lr = 500, 0.5
	for it := 0; it < iters; it++ {
		var ga, gb float64
		for i := range raws {
			p := 1.0 / (1.0 + math.Exp(-(a*raws[i] + b)))
			ga += (p - labels[i]) * raws[i]
			gb += p - labels[i]
		}
		a -= lr * ga / n
		b -= lr * gb / n
	}
	if math.IsNaN(a) || math.IsInf(a, 0) || math.IsNaN(b) || math.IsInf(b, 0) {
		return 1.0, 0.0
	}
	return a, b
}

// FitContentModel fits the like−dislike direction and Platt calibration.
// Fewer than MinLikes liked vectors → a coldstart model (Score always 0; the
// serving layer substitutes gem_score/100, mirroring
// scripts/score_content_preferences.py).
func FitContentModel(liked, disliked []LabeledVec) *ContentModel {
	if len(liked) < MinLikes {
		return &ContentModel{version: "coldstart", a: 1.0, b: 0.0}
	}
	dim := len(liked[0].Vec)
	likeC := make([]float64, dim)
	for _, lv := range liked {
		for i, x := range lv.Vec {
			likeC[i] += x
		}
	}
	for i := range likeC {
		likeC[i] /= float64(len(liked))
	}
	disC := make([]float64, dim)
	if len(disliked) > 0 {
		for _, dv := range disliked {
			for i, x := range dv.Vec {
				disC[i] += x
			}
		}
		for i := range disC {
			disC[i] /= float64(len(disliked))
		}
	}
	diff := make([]float64, dim)
	for i := range diff {
		diff[i] = likeC[i] - disC[i]
	}
	dir := l2norm(diff)

	raws := make([]float64, 0, len(liked)+len(disliked))
	labels := make([]float64, 0, len(liked)+len(disliked))
	for _, lv := range liked {
		raws = append(raws, dot(lv.Vec, dir))
		labels = append(labels, 1)
	}
	for _, dv := range disliked {
		raws = append(raws, dot(dv.Vec, dir))
		labels = append(labels, 0)
	}
	a, b := fitPlatt(raws, labels)

	likedVecs := make([][]float64, len(liked))
	likedNames := make([]string, len(liked))
	for i, lv := range liked {
		likedVecs[i] = lv.Vec
		likedNames[i] = lv.Name
	}
	return &ContentModel{
		version: "content-sim-v1", dir: dir, a: a, b: b,
		likedVecs: likedVecs, likedNames: likedNames,
	}
}

func dot(a, b []float64) float64 {
	n := len(a)
	if len(b) < n {
		n = len(b)
	}
	var s float64
	for i := 0; i < n; i++ {
		s += a[i] * b[i]
	}
	return s
}

// Coldstart reports whether the model has no fitted direction.
func (m *ContentModel) Coldstart() bool { return m.version == "coldstart" }

// Score is the raw taste score: dot(vec, direction); 0 in coldstart.
func (m *ContentModel) Score(vec []float64) float64 {
	if m.Coldstart() {
		return 0
	}
	return dot(vec, m.dir)
}

// PLike maps a raw score through the Platt calibration to a probability.
func (m *ContentModel) PLike(raw float64) float64 {
	return 1.0 / (1.0 + math.Exp(-(m.a*raw + m.b)))
}

// Why names the nearest liked app by dot product (first max wins, as numpy
// argmax); empty in coldstart.
func (m *ContentModel) Why(vec []float64) string {
	if m.Coldstart() || len(m.likedVecs) == 0 {
		return ""
	}
	best, bestI := math.Inf(-1), 0
	for i, lv := range m.likedVecs {
		if s := dot(lv, vec); s > best {
			best, bestI = s, i
		}
	}
	return "similar to " + m.likedNames[bestI]
}

// ParsePgFloat64Array parses a Postgres float array text literal like
// "{0.1,-0.2,0.3}" into float64s (the float32 variant is ParsePgFloatArray;
// the content model runs in float64 for parity with the numpy reference).
func ParsePgFloat64Array(s string) ([]float64, error) {
	s = strings.TrimSpace(s)
	if len(s) < 2 || s[0] != '{' || s[len(s)-1] != '}' {
		return nil, fmt.Errorf("reco: not a pg array literal: %q", s)
	}
	inner := strings.TrimSpace(s[1 : len(s)-1])
	if inner == "" {
		return []float64{}, nil
	}
	parts := strings.Split(inner, ",")
	out := make([]float64, len(parts))
	for i, p := range parts {
		f, err := strconv.ParseFloat(strings.TrimSpace(p), 64)
		if err != nil {
			return nil, fmt.Errorf("reco: bad float %q in pg array", p)
		}
		out[i] = f
	}
	return out, nil
}
```

```go
package repositories

import (
	"sync"
	"time"

	"github.com/edaslakavicius/emerging-app-finder/backend/internal/reco"
	"gorm.io/gorm"
)

// prefEmbCacheTTL bounds how stale the in-process embedding snapshot may get;
// newly embedded apps appear in the queue within this window.
const prefEmbCacheTTL = 10 * time.Minute

// prefEmbCache holds every frontend.app_embeddings row (dim=768) in memory so
// warm review-queue refills do no embedding I/O — the table is ~1k rows and
// immutable per app, so a whole-table snapshot with a short TTL is enough.
// The returned map is a shared snapshot: callers must treat it as read-only
// (refreshes swap in a new map, they never mutate the old one).
var prefEmbCache struct {
	mu       sync.Mutex
	vecs     map[int64][]float64
	loadedAt time.Time
}

func prefEmbeddings(g *gorm.DB) (map[int64][]float64, error) {
	prefEmbCache.mu.Lock()
	defer prefEmbCache.mu.Unlock()
	if prefEmbCache.vecs != nil && time.Since(prefEmbCache.loadedAt) < prefEmbCacheTTL {
		return prefEmbCache.vecs, nil
	}
	var rows []struct {
		AdvertiserID int64  `gorm:"column:advertiser_id"`
		Emb          string `gorm:"column:emb"`
	}
	if err := g.Raw(
		`SELECT advertiser_id, embedding::text AS emb
		 FROM frontend.app_embeddings WHERE dim = 768`,
	).Scan(&rows).Error; err != nil {
		if prefEmbCache.vecs != nil {
			return prefEmbCache.vecs, nil // serve stale over failing the queue
		}
		return nil, err
	}
	m := make(map[int64][]float64, len(rows))
	for _, r := range rows {
		v, err := reco.ParsePgFloat64Array(r.Emb)
		if err != nil {
			return nil, err
		}
		m[r.AdvertiserID] = v
	}
	prefEmbCache.vecs = m
	prefEmbCache.loadedAt = time.Now()
	return m, nil
}
```

```go
// PreferenceQueue returns the user's review queue, scored at serve time so the
// ranking adapts to ratings immediately (no offline preference_scores job in
// the path). Per request it fits the content-preference model from the user's
// like/dislike ratings × cached app embeddings, scores every unrated matview
// candidate, then blends exploit/uncertain/fresh buckets and caps per niche via
// assembleQueue. exclude lists advertiser_ids the client already holds (its
// card buffer / just-rated ids) so background refills never serve duplicates.
// Hero creative + growth are attached only to the final selection, inside the
// same read transaction.
func PreferenceQueue(ctx context.Context, g *gorm.DB, userEmail string, limit int, exclude []int64) ([]models.PreferenceCardRow, error) {
	var outRows []models.PreferenceCardRow
	err := db.WithReadTx(ctx, g, func(tx *gorm.DB) error {
		embs, err := prefEmbeddings(tx)
		if err != nil {
			return err
		}

		// The user's labels, with app names for the "similar to <name>" why.
		// Rated apps may have left the matview, hence the LEFT JOIN + COALESCE
		// (mirrors the Python loader's meta.get(a, ("", 0)) fallback).
		var rated []struct {
			AdvertiserID int64  `gorm:"column:advertiser_id"`
			Rating       string `gorm:"column:rating"`
			AppName      string `gorm:"column:app_name"`
		}
		if err := tx.Raw(
			`SELECT r.advertiser_id, r.rating,
			        COALESCE(m.app_name, m.page_name, '') AS app_name
			 FROM frontend.app_ratings r
			 LEFT JOIN frontend.emerging_scaling_apps m
			   ON m.advertiser_id = r.advertiser_id
			 WHERE r.user_email = ?`, userEmail,
		).Scan(&rated).Error; err != nil {
			return err
		}
		var liked, disliked []reco.LabeledVec
		for _, r := range rated {
			vec, ok := embs[r.AdvertiserID]
			if !ok {
				continue
			}
			lv := reco.LabeledVec{Name: r.AppName, Vec: vec}
			switch r.Rating {
			case "like":
				liked = append(liked, lv)
			case "dislike":
				disliked = append(disliked, lv)
			}
		}
		model := reco.FitContentModel(liked, disliked)

		// Candidates: every unrated matview app, minus the client's exclude list.
		q := `SELECT ` + mvCols + `
		      FROM frontend.emerging_scaling_apps
		      WHERE NOT EXISTS (
		        SELECT 1 FROM frontend.app_ratings r
		        WHERE r.user_email = ?
		          AND r.advertiser_id = frontend.emerging_scaling_apps.advertiser_id)`
		args := []interface{}{userEmail}
		if len(exclude) > 0 {
			q += ` AND frontend.emerging_scaling_apps.advertiser_id NOT IN (?)`
			args = append(args, exclude)
		}
		var scanned []mvScanRow
		if err := tx.Raw(q, args...).Scan(&scanned).Error; err != nil {
			return err
		}

		shortPool := toShortlistRows(scanned)
		cands := make([]prefCandidate, len(shortPool))
		for i, row := range shortPool {
			vec, hasVec := embs[int64(row.AdvertiserID)]
			var p float64
			var why, source string
			switch {
			case !hasVec:
				p, why, source = 0.5, "new", "fresh"
			case model.Coldstart():
				// Serving substitute for an unfitted model: rank by GemScore
				// (mirrors scripts/score_content_preferences.py's coldstart branch).
				p, why, source = row.GemScore/100.0, "", "scored"
			default:
				p, why, source = model.PLike(model.Score(vec)), model.Why(vec), "scored"
			}
			unc := 1.0 - math.Abs(2.0*p-1.0)
			cands[i] = prefCandidate{
				PreferenceCardRow: models.PreferenceCardRow{
					ShortlistRow: row,
					PLike:        p,
					Why:          why,
				},
				Uncertainty: unc,
				Source:      source,
			}
		}

		selected := assembleQueue(cands, queueOpts{
			Limit:         limit,
			PerNicheCap:   defaultPerNicheCap,
			ExploitFrac:   exploitFrac,
			UncertainFrac: uncertainFrac,
			Rng:           rand.New(rand.NewSource(time.Now().UnixNano())),
		})

		shortRows := make([]models.ShortlistRow, len(selected))
		for i := range selected {
			shortRows[i] = selected[i].ShortlistRow
		}

			return err
		}
		outRows = make([]models.PreferenceCardRow, len(selected))

				ShortlistRow: shortRows[i],
				PLike:        selected[i].PLike,
				Why:          selected[i].Why,
			}
		}

	})
	if err != nil {
		return nil, err
	}
	return outRows, nil
}
```

```ts
/**
 * Buffer + refill logic for the review deck. The deck holds a small window of
 * cards; after each rating a debounced background refill fetches a freshly
 * ranked batch (the server rescored with the rating just given) and replaces
 * the unseen tail. Pure module — the React component is a thin consumer.
 */

/** Cards fetched per request — the visible buffer stays about this deep. */
export const REFILL_LIMIT = 12;

/** Keep the current card and this many upcoming cards stable across a refill
 * so the deck never visibly shifts under the user. */
export const KEEP_AHEAD = 1;

/** How many of the most recent session-rated ids ride along in `exclude` —
 * guards the race where a refill lands before the rating row commits. Older
 * ratings are excluded server-side anyway. */
export const RECENT_RATED_SENT = 20;

type HasAdvertiserID = { advertiser_id: number };

/** Replaces the unseen tail of the deck with a freshly ranked batch, keeping
 * everything up to and including the next card after the current index. */
export function mergeRefill<T extends HasAdvertiserID>(
  cards: T[],
  idx: number,
  fresh: T[],
): T[] {
  const kept = cards.slice(0, idx + KEEP_AHEAD + 1);
  const keptIds = new Set(kept.map((c) => c.advertiser_id));
  return [...kept, ...fresh.filter((c) => !keptIds.has(c.advertiser_id))];
}

/** The ids a refill must not serve again: everything still ahead in the deck
 * (including the current card) plus the newest session ratings. */
export function refillExclude<T extends HasAdvertiserID>(
  cards: T[],
  idx: number,
  ratedIds: number[],
): number[] {
  const ids = new Set<number>();
  for (const c of cards.slice(idx)) ids.add(c.advertiser_id);
  for (const id of ratedIds.slice(-RECENT_RATED_SENT)) ids.add(id);
  return [...ids];
}

/** Debounced, single-flight runner: bursts of request() collapse into one run;
 * requests during a run queue exactly one catch-up run; errors are swallowed
 * (a refill is best-effort — the deck keeps serving its buffer). */
export class RefillScheduler {
  private timer: ReturnType<typeof setTimeout> | null = null;
  private running = false;
  private again = false;

  constructor(
    private readonly run: () => Promise<void>,
    private readonly debounceMs = 300,
  ) {}

  request(): void {
    if (this.timer) clearTimeout(this.timer);
    this.timer = setTimeout(() => {
      this.timer = null;
      void this.fire();
    }, this.debounceMs);
  }

  dispose(): void {
    if (this.timer) clearTimeout(this.timer);
    this.timer = null;
  }

  private async fire(): Promise<void> {
    if (this.running) {
      this.again = true;
      return;
    }
    this.running = true;
    try {
      await this.run();
    } catch {
      /* best-effort */
    }
    this.running = false;
    if (this.again) {
      this.again = false;
      void this.fire();
    }
  }
}
```

Replace the file contents with:

```tsx
"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import { useSession } from "next-auth/react";

import { getPreferenceQueue, setRating } from "@/lib/api";
import type { Rating } from "@/lib/api";
import { REFILL_LIMIT, RefillScheduler, mergeRefill, refillExclude } from "@/lib/reviewBuffer";
import type { PreferenceCard } from "@/types/contracts";
import { ReviewCard } from "./ReviewCard";

export function ReviewDeck() {
  const { data: session } = useSession();
  const userEmail = session?.user?.email ?? null;
  const [cards, setCards] = useState<PreferenceCard[]>([]);
  const [idx, setIdx] = useState(0);
  const [loading, setLoading] = useState(true);
  // True once a refill comes back empty: every candidate is rated.
  const [exhausted, setExhausted] = useState(false);
  const ratedIdsRef = useRef<number[]>([]);
  const schedulerRef = useRef<RefillScheduler | null>(null);
  // Refs mirror state for the async refill closure (it must read the values
  // at run time, not the values captured when the scheduler was built).
  const stateRef = useRef({ cards, idx });
  stateRef.current = { cards, idx };

  // Reset the deck when the signed-in user changes. Adjusting state during
  // render (guarded by the change) is React's recommended replacement for
  // calling setState synchronously in an effect — it avoids the cascading-
  // render warning and applies before the children see stale idx/loading.
  const [prevEmail, setPrevEmail] = useState(userEmail);
  if (userEmail !== prevEmail) {
    setPrevEmail(userEmail);
    setIdx(0);
    setLoading(true);
    setExhausted(false);
    ratedIdsRef.current = [];
  }

  useEffect(() => {
    if (!userEmail) return;
    let alive = true;
    // Small initial buffer; background refills keep it topped up and freshly
    // ranked as the user rates (each refill rescored from the latest ratings).
    getPreferenceQueue(userEmail, REFILL_LIMIT)
      .then((c) => {
        if (!alive) return;
        setCards(c);
        setExhausted(c.length === 0);
      })
      .catch(() => {
        if (alive) setCards([]);
      })
      .finally(() => {
        if (alive) setLoading(false);
      });

    const scheduler = new RefillScheduler(async () => {
      const { cards: cur, idx: curIdx } = stateRef.current;
      const exclude = refillExclude(cur, curIdx, ratedIdsRef.current);
      const fresh = await getPreferenceQueue(userEmail, REFILL_LIMIT, exclude);
      setExhausted(fresh.length === 0);
      if (fresh.length > 0) {
        setCards((prev) => mergeRefill(prev, stateRef.current.idx, fresh));
      }
    });
    schedulerRef.current = scheduler;
    return () => {
      alive = false;
      scheduler.dispose();
      schedulerRef.current = null;
    };
  }, [userEmail]);

  const current = cards[idx];

  // If the user rates faster than refills land, the buffer can drain while
  // more candidates exist — nudge another refill instead of dead-ending.
  useEffect(() => {
    if (!loading && !current && !exhausted) schedulerRef.current?.request();
  }, [loading, current, exhausted]);

  const rate = useCallback(
    (rating: Rating, comment = "") => {
      if (!current || !userEmail) return;
      ratedIdsRef.current.push(current.advertiser_id);
      // Optimistic advance; persist in the background, then refill so the next
      // cards reflect the rating just given.
      setIdx((i) => i + 1);
      setRating(current.advertiser_id, userEmail, rating, comment)
        .catch(() => {
          /* best-effort: the card has already advanced; user can re-rate from Discover */
        })
        .finally(() => schedulerRef.current?.request());
    },
    [current, userEmail],
  );

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      const t = e.target as HTMLElement | null;
      if (t && (t.tagName === "TEXTAREA" || t.tagName === "INPUT" || t.isContentEditable)) return;
      // The screenshot lightbox owns the arrow keys while it's open.
      if (document.querySelector("[data-review-lightbox]")) return;
      if (!current) return;
      if (e.key === "ArrowRight") rate("like");
      else if (e.key === "ArrowLeft") rate("dislike");
      else if (e.key === "ArrowDown" || e.key === " ") rate("neutral");
    }
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [current, rate]);

  if (!userEmail) {
    return <p style={{ padding: "var(--sp-8)" }}>Sign in to review.</p>;
  }
  if (loading) {
    return <p style={{ padding: "var(--sp-8)" }}>Loading the queue…</p>;
  }
  if (!current) {
    if (!exhausted) {
      return <p style={{ padding: "var(--sp-8)" }}>Loading more…</p>;
    }
    return (
      <p style={{ padding: "var(--sp-8)" }}>
        All caught up — you’ve reviewed every app in the queue. New ones appear as
        discovery runs.
      </p>
    );
  }

  return <ReviewCard card={current} onRate={rate} />;
}
```
